// PROTOTYPE (bound once): the names whose assigned value is hidden for the
// extent of a call (the value shield), with no scope and no shadow
// declaration.
const hidden = new WeakMap<object, string[][]>();

/** True when the value of `name` is hidden in the engine `ce`. */
export function valueIsHidden(ce: object, name: string): boolean {
  const stack = hidden.get(ce);
  if (stack === undefined) return false;
  for (const names of stack) if (names.includes(name)) return true;
  return false;
}

/** Run `fn` with the values of `names` hidden. */
export function withHiddenValues<T>(
  ce: object,
  names: string[],
  fn: () => T
): T {
  let stack = hidden.get(ce);
  if (stack === undefined) hidden.set(ce, (stack = []));
  stack.push(names);
  try {
    return fn();
  } finally {
    stack.pop();
  }
}
