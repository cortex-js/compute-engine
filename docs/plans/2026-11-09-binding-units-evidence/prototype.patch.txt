+// PROTOTYPE, not to ship: section 7 of
+// `docs/plans/2026-11-09-binding-units-model.md`.
+//
+// A binding unit is one outermost boxing (`ce.parse()`, `ce.box()`,
+// `ce.function()` from raw input). Its environment holds, for each free name
+// (a name that only a boxing declared, with no value and no declared type),
+// the type that the uses in the unit require. A symbol occurrence keeps a
+// reference to the environment of the boxing that made it
+// (`BoxedSymbol._env`).
+//
+// Two environment variables control the prototype:
+// - `CE_BINDING_UNITS=1`: the environment is the answer of
+//   `BoxedSymbol.type` for a free name, and the narrowing write on the shared
+//   definition is not made. Without it (compare mode), the write of today is
+//   made and is the answer, and the environment is built beside it.
+// - `CE_BU_COUNT=1`: each read of the type of a free name computes both
+//   answers and counts the reads where they differ, by read site and by
+//   cause, in `globalThis.__C`.
+
+import { BoxedType } from '../common/type/boxed-type.js';
+
+const ENV: Record<string, string | undefined> =
+  typeof process !== 'undefined' && process.env ? process.env : {};
+
+/** True when the environment is the answer and the shared write is not
+ * made. */
+export const BU_ON = ENV.CE_BINDING_UNITS === '1';
+
+/** Experiment: true when a write in an environment sends no state event. */
+export const BU_NO_EVENT = ENV.CE_BU_NO_EVENT === '1';
+
+/** True when the reads are counted. */
+export const BU_COUNT = ENV.CE_BU_COUNT === '1';
+
+/** The types that one binding unit requires of its free names. The key is
+ * the value definition of the name: a name with no explicit definition is
+ * the same symbol in every unit of a scope, and only its inferred type
+ * belongs to the unit. */
+export class UnitEnvironment {
+  private _types: Map<object, BoxedType> | undefined = undefined;
+  private _scalarUsesMap: Map<object, boolean> | undefined = undefined;
+
+  get(def: object): BoxedType | undefined {
+    return this._types?.get(def);
+  }
+
+  set(def: object, type: BoxedType): void {
+    (this._types ??= new Map()).set(def, type);
+  }
+
+  /** The record that `SCALAR_USES_MAP` (`boxed-symbol.ts`) keeps for a
+   * shared definition, kept here for the unit. */
+  scalarUsesMap(def: object): boolean | undefined {
+    return this._scalarUsesMap?.get(def);
+  }
+
+  setScalarUsesMap(def: object, value: boolean | undefined): void {
+    if (value === undefined) this._scalarUsesMap?.delete(def);
+    else (this._scalarUsesMap ??= new Map()).set(def, value);
+  }

+ */
+export const LEGACY = new UnitEnvironment();
+
+/** Greater than zero while the prototype itself reads the type of a value

+export const internalRead = { depth: 0 };
+
+type Counters = Record<string, number>;
+
+function counters(): Counters {
+  const g = globalThis as { __C?: Counters };
+  return (g.__C ??= {});
+}
+
+export function count(key: string, n = 1): void {
+  const c = counters();
+  c[key] = (c[key] ?? 0) + n;
+}
+
+/**
+ * The function that called the getter in which a read was counted, as
+ * `<file under src>:<class>.<function>`. `skipFile` names the file of the
+ * getter: its frames are skipped (`all`), or only the first one (`one`).
+ */
+function readSite(skipFile: string, skip: 'one' | 'all'): string {
+  const prepare = Error.prepareStackTrace;
+  const limit = Error.stackTraceLimit;
+  Error.stackTraceLimit = 14;
+  Error.prepareStackTrace = (_e, frames) => frames;
+  const holder: { stack?: unknown } = {};
+  Error.captureStackTrace(holder);
+  const frames = holder.stack as NodeJS.CallSite[];
+  Error.prepareStackTrace = prepare;
+  Error.stackTraceLimit = limit;
+  let skipped = 0;
+  for (const frame of frames) {
+    const file = frame.getFileName() ?? '';
+    if (file.includes('binding-unit')) continue;
+    if (file.includes(skipFile) && (skip === 'all' || skipped === 0)) {
+      skipped += 1;
+      continue;
+    }
+    const name =
+      frame.getFunctionName() ?? frame.getMethodName() ?? '<anonymous>';
+    return `${file.replace(/^.*\/(src|test)\//, '$1/')}:${name}`;
+  }
+  return '<top>';
+}
+
+/** The relation of the answer of the environment to the answer of today. */
+function relation(envType: BoxedType, today: BoxedType): string {
+  if (envType.isUnknown) return 'env-unknown';
+  if (today.type === 'never') return 'today-never';
+  if (today.isUnknown) return 'today-unknown';
+  if (today.matches(envType)) return 'env-wider';
+  if (envType.matches(today)) return 'env-narrower';
+  return 'unrelated';
+}
+
+/**
+ * Count a read of the type of a free name through a symbol occurrence.
+ * `envType` is the answer of the environment, `today` the answer of the
+ * shared definition, `hasEnv` whether the occurrence has an environment
+ * (false for a symbol made from a name alone, outside a boxing).
+ */
+export function countSymbolRead(
+  envType: BoxedType,
+  today: BoxedType,
+  hasEnv: boolean
+): void {
+  count('reads/symbol/total');
+  if (
+    envType.type === today.type ||
+    envType.toString() === today.toString()
+  )
+    return;
+  const cause = !hasEnv
+    ? 'no-environment'
+    : `environment/${relation(envType, today)}`;
+  count(`diff/symbol/${cause}`);
+  count(`site/symbol/${cause}/${readSite('boxed-symbol', 'one')}`);
+}
+
+/** Count a read of the type of a free name on its value definition, with no
+ * symbol occurrence at hand. The answer with the switch on is `unknown`. */
+export function countDirectRead(today: BoxedType): void {
+  count('reads/direct/total');
+  if (today.isUnknown) return;
+  count('diff/direct');
+  count(`site/direct/${readSite('boxed-value-definition', 'all')}`);
+}
+
+/** Take the counters and reset them. */
+export function takeCounters(): Counters | undefined {
+  const g = globalThis as { __C?: Counters };
+  const c = g.__C;
+  if (c === undefined) return undefined;
+  g.__C = undefined;
+  for (const _k in c) return c;
+  return undefined;
+}

+import type { Expression as MathJson } from '../../src/math-json/types';
+
+// A use of a free name narrows the inferred type of its definition. For a
+// name of a scope that existed before the boxing (a name of the host), the
+// definition keeps one requirement per boxing, and a boxing that the host
+// starts reads the type without the requirements of earlier boxings
+// (`src/compute-engine/host-boxing.ts`). So the result of boxing an
+// expression does not depend on the expressions boxed before it, and the
+// type that a name reports is the meet of the requirements, or `unknown`
+// when two requirements conflict.
+// (`docs/plans/2026-11-09-inference-as-a-result.md`, step 1.)
+
+type Input = string | MathJson;
+
+function box(ce: ComputeEngine, input: Input) {
+  return typeof input === 'string' ? ce.parse(input) : ce.box(input);
+}
+
+function typeOf(ce: ComputeEngine, name: string): string {
+  return ce.box(name).type.toString();
+}
+
+describe('A READING DOES NOT DEPEND ON AN EARLIER EXPRESSION', () => {
+  test('`x(3)` after `x + 1` is a call', () => {
+    const ce = new ComputeEngine();
+    expect(ce.parse('x + 1').json).toEqual(['Add', 'x', 1]);
+    expect(ce.parse('x(3)').json).toEqual(['x', 3]);
+  });
+
+  test('`x + 1` after `x(3)` is a sum, not an error', () => {
+    const ce = new ComputeEngine();
+    expect(ce.parse('x(3)').json).toEqual(['x', 3]);
+    const sum = ce.parse('x + 1');
+    expect(sum.json).toEqual(['Add', 'x', 1]);
+    expect(sum.isValid).toBe(true);
+  });
+
+  test('a speculative parse reads as the normal parse', () => {
+    const ce = new ComputeEngine();
+    ce.parse('x + 1');
+    expect(ce.parse('x(3)', { speculative: true }).json).toEqual(['x', 3]);
+    // The speculative parse leaves no requirement.
+    expect(typeOf(ce, 'x')).toBe('number');
+  });
+
+  test('the name stays declared after the first parse', () => {
+    const ce = new ComputeEngine();
+    ce.parse('x + 1');
+    expect(ce.lookupDefinition('x')).toBeDefined();
+    expect(typeOf(ce, 'x')).toBe('number');
+  });
+
+  test('two conflicting requirements give the type `unknown`, in either order', () => {
+    for (const order of [
+      ['x + 1', 'x(3)'],
+      ['x(3)', 'x + 1'],
+    ]) {
+      const ce = new ComputeEngine();
+      for (const latex of order) ce.parse(latex);
+      expect(typeOf(ce, 'x')).toBe('unknown');
+    }
+  });
+
+  test('requirements that agree give their meet', () => {
+    const ce = new ComputeEngine();
+    ce.parse('x + 1');
+    ce.parse('\\sin(x)');
+    expect(typeOf(ce, 'x')).toBe('number');
+  });
+
+  test('a numeric use that maps and a collection use give the collection', () => {
+    // The scalar exception of the narrowing (`narrowInferredType()`): `x + 1`
+    // maps over a numeric collection, so it does not prove that `x` is a
+    // scalar.
+    for (const order of [
+      [['Add', 'v', 1], ['Add', ['At', 'v', 1], ['At', 'v', 2]]],
+      [['Add', ['At', 'v', 1], ['At', 'v', 2]], ['Add', 'v', 1]],
+    ] as MathJson[][]) {
+      const ce = new ComputeEngine();
+      for (const input of order) ce.box(input);
+      expect(typeOf(ce, 'v')).toBe('indexed_collection<number>');
+    }
+  });
+
+  test('a numeric use that does not map and a collection use conflict', () => {
+    for (const order of [
+      [['Range', 'n'], ['At', 'n', 1]],
+      [['At', 'n', 1], ['Range', 'n']],
+    ] as MathJson[][]) {
+      const ce = new ComputeEngine();
+      for (const input of order) ce.box(input);
+      expect(typeOf(ce, 'n')).toBe('unknown');
+    }
+  });
+
+  test('the call of a library operator after a numeric use of its name', () => {
+    // `N + 1` declares a variable `N`; `N(3.14159, 2)` still calls the
+    // library operator, as it does in a fresh engine.
+    const ce = new ComputeEngine();
+    ce.parse('N + 1');
+    expect(ce.parse('N(3.14159, 2)').evaluate().json).toBe(3.1);
+    ce.parse('x + N');
+    expect(ce.parse('N(2.3)').evaluate().json).toBe(2.3);
+  });
+
+  test('a boxing inside an evaluation reads the reported type', () => {
+    // The evaluation of an expression boxed earlier is not a boxing of the
+    // host: it keeps the type that its own boxing inferred.
+    const ce = new ComputeEngine();
+    const expr = ce.parse('x^2 + 2x');
+    ce.parse('x(3)');
+    expect(expr.subs({ x: 3 }).evaluate().json).toBe(15);
+  });
+});
+
+describe('AN INFERRED TYPE DOES NOT REFUSE AN ASSIGNMENT', () => {
+  test('after one requirement', () => {
+    const ce = new ComputeEngine();
+    ce.parse('x + 1');
+    ce.assign('x', ce.string('hello'));
+    expect(ce.box('x').evaluate().json).toBe("'hello'");
+  });
+
+  test('after two conflicting requirements', () => {
+    const ce = new ComputeEngine();
+    ce.parse('x + 1');
+    ce.parse('x(3)');
+    ce.assign('x', 5);
+    expect(ce.parse('x + 1').evaluate().json).toBe(6);
+  });
+
+  test('a declaration replaces the inferred type', () => {
+    const ce = new ComputeEngine();
+    ce.parse('x + 1');
+    ce.declare('x', 'string');
+    expect(typeOf(ce, 'x')).toBe('string');
+    // The requirements of the earlier boxings no longer apply.
+    expect(ce.parse('x').type.toString()).toBe('string');
+  });
+});
+
+describe('ROLLBACK AND CHECKPOINT', () => {
+  test('a rollback frame removes the requirements its boxings added', () => {
+    const ce = new ComputeEngine();
+    ce.parse('x + 1');
+    ce._withBoxingPassWindow(() =>
+      ce._withRolledBackInference(() => {
+        ce.parse('x(3)');
+        expect(typeOf(ce, 'x')).toBe('unknown');
+      })
+    );
+    expect(typeOf(ce, 'x')).toBe('number');
+    // The requirement of `x + 1` is still the only one: a boxing that
+    // agrees with it keeps the type.
+    ce.parse('2x');
+    expect(typeOf(ce, 'x')).toBe('number');
+  });
+
+  test('a checkpoint restore removes the requirements', () => {
+    const ce = new ComputeEngine();
+    ce.parse('x + 1');
+    const cp = ce.checkpoint();
+    ce.parse('x(3)');
+    expect(typeOf(ce, 'x')).toBe('unknown');
+    ce.restore(cp);
+    expect(typeOf(ce, 'x')).toBe('number');
+    ce.parse('\\cos(x)');
+    expect(typeOf(ce, 'x')).toBe('number');
+  });
+});

+  test.each(PAIRS.map(([a, b]) => [JSON.stringify(a), JSON.stringify(b), a, b]))(
+    '%s and %s',
+    (_a, _b, a, b) => {
+      const first = new ComputeEngine();
+      const a1 = box(first, a as Input);
+      const b1 = box(first, b as Input);
+      const second = new ComputeEngine();
+      const b2 = box(second, b as Input);
+      const a2 = box(second, a as Input);
+      expect(a1.json).toEqual(a2.json);
+      expect(b1.json).toEqual(b2.json);
+      const names = [
+        ...new Set([...a1.unknowns, ...b1.unknowns, ...a2.unknowns]),
+      ].sort();
+      expect(names.map((n) => [n, typeOf(first, n)])).toEqual(
+        names.map((n) => [n, typeOf(second, n)])

+    }
+  );
